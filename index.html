    <h1>🐋 Калькулятор китов</h1>

    <p>Модель: u(t+1) = u(t) × exp(r × (1 − u(t−1)))</p>

    

    <label>Параметр r: <span id="rValue">1.0</span></label><br>

    <input type="range" id="rSlider" min="0.5" max="1.8" step="0.01" value="1.0" style="width: 80%;">

    <button onclick="simulate()">Рассчитать</button>

    

    <canvas id="whaleCanvas" width="200" height="100"></canvas>

    

    <div id="result"></div>

    

    <canvas id="graphCanvas" width="500" height="200"></canvas>

<script>

    function ricerModel(r, u0, u1, steps) {

        let u = [u0, u1];

        for (let i = 2; i <= steps; i++) {

            u.push(u[u.length-1] * Math.exp(r * (1 - u[u.length-2])));

        }

        return u;

    }

    

    function drawWhale(size, isStable) {

        let canvas = document.getElementById('whaleCanvas');

        let ctx = canvas.getContext('2d');

        ctx.clearRect(0, 0, canvas.width, canvas.height);

        ctx.fillStyle = isStable ? "#2e7d32" : "#b71c1c";

        ctx.fillRect(20, 30, 30 + size*30, 40);

        ctx.fillStyle = "white";

        ctx.fillRect(25, 20, 10, 10);

        ctx.fillStyle = "black";

        ctx.fillRect(28, 22, 5, 5);

        ctx.fillStyle = "black";

        ctx.font = "20px Arial";

        ctx.fillText(isStable ? "😊" : "😢", 100, 60);

    }

    

    function drawGraph(data, r) {

        let canvas = document.getElementById('graphCanvas');

        let ctx = canvas.getContext('2d');

        let w = canvas.width, h = canvas.height;

        ctx.clearRect(0, 0, w, h);

        let maxVal = Math.max(...data);

        let scaleY = (h-40)/maxVal;

        ctx.beginPath();

        for (let i=1; i<data.length; i++) {

            let x1 = (i-1)*w/(data.length-1);

            let y1 = h-20 - data[i-1]*scaleY;

            let x2 = i*w/(data.length-1);

            let y2 = h-20 - data[i]*scaleY;

            ctx.moveTo(x1, y1);

            ctx.lineTo(x2, y2);

            ctx.stroke();

        }

        ctx.fillText("r = "+r.toFixed(2), 10, 20);

    }

    

    function simulate() {

        let r = parseFloat(document.getElementById('rSlider').value);

        document.getElementById('rValue').innerText = r;

        let data = ricerModel(r, 0.2, 0.3, 80);

        let last = data[data.length-1];

        let isStable = (last > 0.1 && last < 1.5 && r < 1.0);

        document.getElementById('result').innerHTML = "<strong>Численность: "+last.toFixed(3)+"</strong><br>"+(isStable ? "✅ Устойчиво" : "⚠️ Неустойчиво/колебания");

        document.getElementById('result').className = isStable ? "stable" : "unstable";

        drawWhale(Math.min(1.2, Math.max(0.2, last)), isStable);

        drawGraph(data, r);

    }

    

    window.onload = simulate;

</script>
